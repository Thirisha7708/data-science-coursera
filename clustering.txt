# Load dataset
data(iris)

# Select only numeric features
iris_data <- iris[, 1:4]

# Standardize the data
iris_scaled <- scale(iris_data)

# Apply K-Means clustering (K = 3)
set.seed(123)
kmeans_result <- kmeans(iris_scaled, centers = 3, nstart = 25)

# Confusion matrix
conf_matrix <- table(iris$Species, kmeans_result$cluster)

print("Confusion Matrix:")
print(conf_matrix)

# Calculate accuracy correctly
accuracy <- sum(apply(conf_matrix, 1, max)) / sum(conf_matrix)

print("Accuracy:")
print(accuracy)
